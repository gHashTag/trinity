## 2. CAPO Oracle Staleness Handling

### v3.0 Vulnerability
```solidity
// CATASTROPHIC: Stale oracle = skip validation!
(uint256 twapPrice, bool valid) = oracle.getTwiPrice(address(asset));
if (!valid) return; // ← Vulnerability!
```

### v3.1 Fix: Strict Oracle Validation
```solidity
uint256 public constant STALENESS_THRESHOLD = 300; // 5 minutes
uint256 public constant MAX_YEARLY_RATIO_GROWTH_BPS = 800; // 8%
uint256 public constant MAX_STALE_ORACLE_DEPOSIT = 1000 * 1e18;

function _validateCapo(uint256 assets) internal view {
    if (address(oracle) == address(0)) {
        require(assets <= MAX_STALE_ORACLE_DEPOSIT, "Oracle required");
        return;
    }

    (uint256 twapPrice, bool valid, uint256 timestamp) =
        oracle.getTwiPriceWithTimestamp(address(asset));

    require(valid, "Oracle stale");
    require(block.timestamp - timestamp < STALENESS_THRESHOLD, "Price too old");

    // Check yearly growth for correlated assets (wstETH pattern)
    (uint256 refPrice, uint256 refTime) = oracle.getReferencePrice(address(asset));
    if (refPrice > 0 && refTime > 0) {
        uint256 growthBps = _calculateYearlyGrowth(twapPrice, refPrice, refTime);
        require(growthBps <= MAX_YEARLY_RATIO_GROWTH_BPS, "Excessive growth");
    }

    // ... existing deviation logic ...
}

function _calculateYearlyGrowth(
    uint256 currentPrice,
    uint256 referencePrice,
    uint256 referenceTime
) internal view returns (uint256) {
    if (referencePrice == 0) return 0;

    uint256 timeElapsed = block.timestamp - referenceTime;
    if (timeElapsed == 0) return 0;

    // Extrapolate to yearly basis
    uint256 priceIncrease = currentPrice > referencePrice ?
        currentPrice - referencePrice : 0;
    uint256 yearlyBps = (priceIncrease * 365 days * 10000) /
                       (referencePrice * timeElapsed);
    return yearlyBps;
}
```

## 4. Adaptive Circuit Breaker

### v3.0 Vulnerability
```solidity
// Fixed 20% threshold - too slow for cascade conditions[7]
uint256 public constant CIRCUIT_BREAKER_THRESHOLD_BPS = 2000;
```

### v3.1 Fix: Leverage-Adaptive Threshold
```solidity
uint256 public constant BASE_THRESHOLD_BPS = 2000; // 20%
uint256 public constant TIGHT_THRESHOLD_BPS = 1000; // 10%
uint256 public constant HIGH_LEVERAGE_BPS = 8000; // 80% util

uint256 public circuitBreakerThresholdBps = BASE_THRESHOLD_BPS;
uint256 public lastLeverageCheckTime;

function updateCircuitBreakerThreshold() internal {
    if (block.timestamp >= lastLeverageCheckTime + 1 hours) {
        uint256 utilization = (_managedAssets * 10000) / maxCapacity;

        // Tighten threshold when leverage is high
        circuitBreakerThresholdBps = utilization >= HIGH_LEVERAGE_BPS ?
            TIGHT_THRESHOLD_BPS : BASE_THRESHOLD_BPS;

        lastLeverageCheckTime = block.timestamp;
    }
}

function triggerCircuitBreaker(uint256 observedPrice) external onlyOwner {
    updateCircuitBreakerThreshold();

    uint256 deviationBps = lastSnapshotPrice > 0 ?
        _calculateDeviationBps(observedPrice, lastSnapshotPrice) : 0;

    if (deviationBps >= circuitBreakerThresholdBps) {
        uint256 duration = _escalatingDuration(deviationBps);
        circuitBreakerUntil = block.timestamp + duration;
        _pause();

        emit CircuitBreakerTripped(lastSnapshotPrice, observedPrice, duration);
    }

    lastSnapshotPrice = observedPrice;
    lastSnapshotTime = block.timestamp;
}

function _escalatingDuration(uint256 deviationBps) internal pure returns (uint256) {
    if (deviationBps >= 3000) return 4 hours;  // >30%
    if (deviationBps >= 2000) return 2 hours;  // >20%
    return 1 hours;
}
```

## 6. Dynamic Deterrence Calculation

### v3.0 Vulnerability
```zig
// Fixed 95% - doesn't adapt to conditions[29]
pub const VERIFICATION_PROBABILITY: f64 = 0.95;
```

### v3.1 Fix: Coverage-Based Probability
```zig
pub const MonitoringCoverage = struct {
    coverage_ratio: f64 = 0.95,        // % of nodes monitored
    independent_monitors: u32 = 3,     // Independent sources
    last_update: i64,
};

pub fn calculateDynamicPVerified(
    self: *TokenStakingEngine,
    node_id: [32]u8,
    level: SlashLevel
) !f64 {
    const coverage = self.monitoring_coverage;
    var p = coverage.coverage_ratio;

    // Bonus for independent monitors
    p += @as(f64, @floatFromInt(coverage.independent_monitors)) * 0.01;

    // Penalty for low-reputation nodes
    if (self.reputation_engine.getScore(node_id)) |score| {
        if (score < 0.3) p -= 0.05;
    }

    // Correlation bonus (mass slash = higher detection)
    const recent_slashes = self.getRecentSlashCount(1 days);
    if (recent_slashes > 10) p += 0.03;

    // Clamp to [0.5, 0.99]
    return @max(0.5, @min(0.99, p));
}

pub fn slashWithDynamicDeterrence(
    self: *TokenStakingEngine,
    node_id: [32]u8,
    level: SlashLevel,
    estimated_gain_wei: u128,
    reason: []const u8
) !u128 {
    const entry = self.stakes.get(node_id) orelse return error.NotStaked;
    const remaining = entry.staked_wei - entry.slashed_wei;

    const p_verified = try self.calculateDynamicPVerified(node_id, level);
    const expected_loss = @as(f64, @floatFromInt(remaining)) * p_verified * level.rate();
    const expected_loss_wei = @intFromFloat(expected_loss);

    if (expected_loss_wei <= estimated_gain_wei) {
        std.log.warn("Slash INSUFFICIENT: stake={d}, gain={d}, p={d:.2}",
            .{remaining, estimated_gain_wei, p_verified});
        return error.InsufficientDeterrence;
    }

    return self.slash(node_id, level, reason);
}
```

## 7. Systemic Cascade Protection

### v3.0 Vulnerability
```solidity
// NO inter-protocol protection - October 2025 scenario repeats
```

### v3.1 Fix: Cascade Detection
```solidity
uint256 public constant SYSTEMIC_LEVERAGE_THRESHOLD_BPS = 8500;
uint256 public constant CASCADE_WINDOW = 5 minutes;
uint256 public constant CASCADE_TRIGGER_BPS = 500; // 5% of TVD

mapping(uint256 => uint256) public liquidationTimestamps;
uint256 public lastCascadeCheck;

event SystemicRiskAlert(uint256 liquidations, uint256 window);

function checkSystemicCascade() internal {
    uint256 windowStart = block.timestamp - CASCADE_WINDOW;
    uint256 windowLiquidations = 0;

    for (uint256 t = lastCascadeCheck; t <= block.timestamp; t++) {
        windowLiquidations += liquidationTimestamps[t];
    }

    uint256 threshold = (_managedAssets * CASCADE_TRIGGER_BPS) / 10000;
    if (windowLiquidations > threshold) {
        emit SystemicRiskAlert(windowLiquidations, CASCADE_WINDOW);

        // Auto-pause + extended circuit breaker
        _pause();
        circuitBreakerUntil = block.timestamp + 4 hours;
    }

    lastCascadeCheck = block.timestamp;
}

function recordLiquidation(uint256 amount) internal {
    liquidationTimestamps[block.timestamp] += amount;
    checkSystemicCascade();
}
```
