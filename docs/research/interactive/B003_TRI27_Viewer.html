TRI-27 MOV Algorithm

TRI27_MOV(dst, src)
procedure TRI27_MOV(dst, src)
// Check bank permissions
if isKernelBank(src) then
error(Cross-bank from kernel not allowed);
end if
// Get source trit
src_trit ← READ_TRIT(src);
// Write destination trit
if isUserBank(dst) then
WRITE_TRIT(dst, src_trit);
end if
return;
end procedure
