## Part 1: The Reproducibility Spectrum

### 1.1 Definitions

| Level | Definition | Example |
|-------|------------|---------|
| **Repeatability** | Same team, same data, same code | Re-running own script |
| **Replicability** | Different team, same data, same code | Published code + data |
| **Reproducibility** | Different team, different data, same method | Re-implementation |
| **Generalizability** | Different method, same question | Meta-analysis |

### 1.2 Target: Replicability

For this guide, we target **replicability**: others can reproduce your results using your code and data.

## Part 4: Random Seed Management

### 4.1 Hierarchical Seeding

```python
import random
import numpy as np
import torch

def set_all_seeds(seed: int):
    """
    Set all random seeds for reproducibility.

    Args:
        seed: Random seed (integer)
    """
    random.seed(seed)
    np.random.seed(seed)
    torch.manual_seed(seed)

    if torch.cuda.is_available():
        torch.cuda.manual_seed(seed)
        torch.cuda.manual_seed_all(seed)

    # Ensure deterministic CUDA operations
    torch.backends.cudnn.deterministic = True
    torch.backends.cudnn.benchmark = False

# Usage in experiment
set_all_seeds(config["experiment"]["seed"])
```

### 4.2 Seed from Configuration

```python
import hashlib

def seed_from_config(config: dict) -> int:
    """
    Generate deterministic seed from config hash.

    This ensures that the same config always produces the same seed,
    while different configs produce different seeds.
    """
    config_str = str(sorted(config.items()))
    hash_int = int(hashlib.sha256(config_str.encode()).hexdigest(), 16)
    return hash_int % (2 ** 32)

# Usage
seed = seed_from_config(config)
set_all_seeds(seed)
```

### 5.2 Data Versioning

```python
import hashlib
from pathlib import Path

def compute_checksum(filepath: Path) -> str:
    """Compute SHA256 checksum of file."""
    sha256 = hashlib.sha256()
    with open(filepath, 'rb') as f:
        for chunk in iter(lambda: f.read(4096), b''):
            sha256.update(chunk)
    return sha256.hexdigest()

def verify_data_integrity(manifest: dict) -> bool:
    """
    Verify data files against checksum manifest.

    Args:
        manifest: Dict mapping filenames to expected checksums

    Returns:
        True if all files match
    """
    for filename, expected_checksum in manifest.items():
        filepath = Path(filename)
        if not filepath.exists():
            print(f"Missing: {filename}")
            return False

        actual_checksum = compute_checksum(filepath)
        if actual_checksum != expected_checksum:
            print(f"Mismatch: {filename}")
            print(f"  Expected: {expected_checksum}")
            print(f"  Actual:   {actual_checksum}")
            return False

    return True
```

## Part 6: Experiment Tracking

### 6.1 Experiment Logging

```python
import json
import logging
from datetime import datetime
from pathlib import Path

class ExperimentLogger:
    """Comprehensive experiment logging."""

    def __init__(self, log_dir: str, experiment_name: str):
        self.log_dir = Path(log_dir)
        self.experiment_name = experiment_name
        self.start_time = datetime.now()

        # Create experiment directory
        self.exp_dir = self.log_dir / experiment_name
        self.exp_dir.mkdir(parents=True, exist_ok=True)

        # Setup logging
        self._setup_logging()

        # Log configuration
        self.log_config()

    def _setup_logging(self):
        """Setup file and console logging."""
        log_file = self.exp_dir / "experiment.log"

        # Create logger
        self.logger = logging.getLogger(self.experiment_name)
        self.logger.setLevel(logging.INFO)

        # File handler
        fh = logging.FileHandler(log_file)
        fh.setLevel(logging.INFO)

        # Console handler
        ch = logging.StreamHandler()
        ch.setLevel(logging.INFO)

        # Formatter
        formatter = logging.Formatter(
            '%(asctime)s - %(name)s - %(levelname)s - %(message)s'
        )
        fh.setFormatter(formatter)
        ch.setFormatter(formatter)

        self.logger.addHandler(fh)
        self.logger.addHandler(ch)

    def log_config(self, config: dict):
        """Log experiment configuration."""
        config_file = self.exp_dir / "config.json"
        with open(config_file, 'w') as f:
            json.dump(config, f, indent=2)

        self.logger.info(f"Config saved to {config_file}")

    def log_metrics(self, metrics: dict, step: int):
        """Log metrics at a given step."""
        metrics_file = self.exp_dir / "metrics.jsonl"

        with open(metrics_file, 'a') as f:
            record = {
                "step": step,
                "timestamp": datetime.now().isoformat(),
                **metrics
            }
            f.write(json.dumps(record) + '\n')

        self.logger.info(f"Step {step}: {metrics}")

    def log_artifact(self, artifact_path: str, artifact_type: str):
        """Log an artifact (model checkpoint, figure, etc.)."""
        import shutil

        artifact_dir = self.exp_dir / "artifacts" / artifact_type
        artifact_dir.mkdir(parents=True, exist_ok=True)

        dest = artifact_dir / Path(artifact_path).name
        shutil.copy2(artifact_path, dest)

        self.logger.info(f"Artifact saved: {dest}")

    def finalize(self):
        """Finalize experiment log."""
        end_time = datetime.now()
        duration = (end_time - self.start_time).total_seconds()

        summary = {
            "experiment_name": self.experiment_name,
            "start_time": self.start_time.isoformat(),
            "end_time": end_time.isoformat(),
            "duration_seconds": duration,
            "status": "completed"
        }

        summary_file = self.exp_dir / "summary.json"
        with open(summary_file, 'w') as f:
            json.dump(summary, f, indent=2)

        self.logger.info(f"Experiment completed in {duration:.2f}s")
```

## Part 7: Testing for Reproducibility

### 7.1 Unit Tests

```python
import pytest
import numpy as np

def test_brier_score_deterministic():
    """Test that Brier score is deterministic with fixed seed."""
    from src.metrics import brier_score

    # Set seed
    np.random.seed(42)

    # Generate test data
    y_true = np.random.randint(0, 2, size=100)
    y_prob = np.random.random(100)

    # Compute score
    score1 = brier_score(y_true, y_prob)

    # Reset seed and recompute
    np.random.seed(42)
    y_true2 = np.random.randint(0, 2, size=100)
    y_prob2 = np.random.random(100)
    score2 = brier_score(y_true2, y_prob2)

    # Should be identical
    assert score1 == score2
    assert np.array_equal(y_true, y_true2)
    assert np.array_equal(y_prob, y_prob2)
```

### 7.2 Integration Tests

@pytest.mark.integration
def test_full_pipeline_reproducible():
